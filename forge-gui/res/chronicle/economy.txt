# Chronicle economy knobs. All values are D5-numbers-pass tuning surface.
# Currency is period dollars, integer cents.

# Free daily ration: packs granted at collection (the played-day tick).
rationPacks=2

# Allowance stipend: weekly lump on the played-day schedule, first payday on
# day 0 so the day-one starter purchase (895) is affordable. The MVP's only
# income source - see ADR-0029.
stipendCents=1000
stipendPeriodDays=7

# LGS daily deal slots (the seeded stock roll).
lgsStockSlots=4

# Kitchen-table purse: base cents a rival's win pays, scaled by that rival's
# pursePercent in rivals.txt. Each rival pays at most once per played day;
# rematches are free to play and pay nothing (ADR-0071). Seeded so beating the
# mid-tier kid is worth roughly half a booster - the sink should reward showing
# up with a real deck, not replace the allowance.
pursebaseCents=150

# Playing for keeps: a rival stops putting cards up once their collection is
# down to about this many. Their deck needs ~36 non-basics, so this leaves a
# real margin of spares - a kid who has been cleaned out says so rather than
# handing over their last playable. This is the ONLY limit on ante games; the
# channel is otherwise unbounded and priced in risk (ADR-0071 + the 08-22
# two-speed amendment).
anteRivalFloorCards=75

# Buylist base (what the shop PAYS), cents per rarity. Notables multiply
# these - see notables.txt.
buylistBasicLandCents=1
buylistCommonCents=2
buylistUncommonCents=10
buylistRareCents=40
